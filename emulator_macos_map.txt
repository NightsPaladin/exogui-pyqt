# emulator_macos_map.txt — Windows/Linux to macOS emulator substitution table
#
# Maps emulator commands from eXo collection Windows (.bat) and Linux (.bsh)
# scripts to their nearest macOS native equivalent.
#
# Intended use: when no dosbox_macos.txt entry exists for a game, look up the
# Linux command from dosbox_linux.txt (or the Windows command from dosbox.txt /
# appleIIGS.txt) here to determine the correct macOS emulator to invoke.
#
# Format:  <linux_or_windows_command> = <macos_emulator_name>
#
# <macos_emulator_name> must match an EmulatorSpec.name in emulator_deps.py
# (e.g. "dosbox-staging", "dosbox-x", "scummvm") or be a bare binary name
# resolvable via PATH.  Use "unavailable" when no macOS equivalent exists.
#
# ── eXoDOS · eXoWin3x · eXoDemoScene — DOSBox (Linux flatpak) ───────────────
flatpak run com.retro_exo.dosbox-074r3-1                                   = dosbox-staging
flatpak run com.retro_exo.dosbox-ece-r4301                                 = dosbox-staging
flatpak run com.retro_exo.dosbox-ece-r4358                                 = dosbox-staging
flatpak run com.retro_exo.dosbox-ece-r4482                                 = dosbox-staging
flatpak run com.retro_exo.dosbox-staging-081-2                             = dosbox-staging
flatpak run com.retro_exo.dosbox-staging-082-0                             = dosbox-staging
flatpak run com.retro_exo.dosbox-x-08220                                   = dosbox-x
flatpak run com.retro_exo.dosbox-x-20240701                                = dosbox-x
#
# DOSBox variants running under Wine on Linux — map to native macOS builds.
# Daum → dosbox-x (closest feature parity); vanilla/ECE/GunStick → staging.
flatpak run com.retro_exo.wine emulators/dosbox/DOSBox.exe                 = dosbox-staging
flatpak run com.retro_exo.wine emulators/dosbox/daum/dosbox.exe            = dosbox-x
flatpak run com.retro_exo.wine emulators/dosbox/GunStick_dosbox/dosbox.exe = dosbox-staging
flatpak run com.retro_exo.wine emulators/dosbox/ece4230/dosbox.exe         = dosbox-staging
flatpak run com.retro_exo.wine emulators/dosbox/ece4481/dosbox.exe         = dosbox-staging

# ── eXoAppleIIGS — GSplus / GSport / MAME (Windows) ─────────────────────────
# GSport is a legacy fork; mapped to gsplus (nearest available macOS emulator).
# KEGS is the recommended modern Apple IIGS emulator for macOS; requires gsplus
# or kegs to be installed — the GUI will prefer whichever is detected first.
GSPlus\gsplus.exe                                                           = gsplus
GSPort\gsport.exe                                                           = gsplus
MAME                                                                        = mame

# DREAMM is cross-platform; macOS build: https://dreamm.aarongiles.com/

dreamm                                                                      = dreamm
